cmake_minimum_required(VERSION 3.16)
project(RishaEngine VERSION 0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
 set(CMAKE_BUILD_TYPE Release)
endif()
option(RISHA_BUILD_TESTS "Build Risha Engine tests" ON)

add_library(risha_core STATIC
 src/core/Time.cpp src/core/Logger.cpp src/core/Text.cpp src/core/Project.cpp
 src/script/Script.cpp src/editor/Scene.cpp src/runtime/Runtime.cpp
 src/gfx/Canvas.cpp src/gfx/Font.cpp src/gfx/FontData.cpp src/gfx/Text.cpp
 src/ui/Feather.cpp src/ui/Ui.cpp)
target_include_directories(risha_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(risha_core PUBLIC cxx_std_17)

set(RISHA_PLATFORM_SOURCES)
if(WIN32)
 list(APPEND RISHA_PLATFORM_SOURCES src/platform/Win32Window.cpp)
else()
 find_package(X11 REQUIRED)
 list(APPEND RISHA_PLATFORM_SOURCES src/platform/X11Window.cpp)
endif()

add_executable(risha_editor src/main.cpp src/app/App.cpp src/app/Game.cpp ${RISHA_PLATFORM_SOURCES})
target_include_directories(risha_editor PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/src/app)
target_link_libraries(risha_editor PRIVATE risha_core)
if(WIN32)
 target_link_libraries(risha_editor PRIVATE user32 gdi32 shell32 ole32)
else()
 target_link_libraries(risha_editor PRIVATE ${X11_LIBRARIES})
 target_include_directories(risha_editor PRIVATE ${X11_INCLUDE_DIR})
endif()

if(MSVC)
 foreach(target risha_core risha_editor)
  target_compile_options(${target} PRIVATE /W4 /utf-8)
 endforeach()
else()
 foreach(target risha_core risha_editor)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endforeach()
 set(CMAKE_CXX_FLAGS_RELEASE "-Os")
 set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
endif()

if(RISHA_BUILD_TESTS)
 enable_testing()
 foreach(name core script scene runtime text demo)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/${name}_tests.cpp)
   add_executable(risha_${name}_tests tests/${name}_tests.cpp)
   target_link_libraries(risha_${name}_tests PRIVATE risha_core)
   target_compile_definitions(risha_${name}_tests PRIVATE RISHA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
   if(MSVC)
    target_compile_options(risha_${name}_tests PRIVATE /W4 /utf-8)
   else()
    target_compile_options(risha_${name}_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
   endif()
   add_test(NAME risha_${name}_tests COMMAND risha_${name}_tests)
  endif()
 endforeach()
endif()

install(TARGETS risha_editor RUNTIME DESTINATION bin)
install(DIRECTORY examples/ DESTINATION share/risha/examples)
install(DIRECTORY docs/ DESTINATION share/risha/docs)
install(FILES risha.desktop DESTINATION share/applications)
